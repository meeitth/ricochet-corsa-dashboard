        function handleLogin(e) {
            e.preventDefault();
            const uInput = document.getElementById('loginUsername').value.trim();
            const pInput = document.getElementById('loginPassword').value.trim();
            const errBox = document.getElementById('loginErrorMsg');

            const foundUser = usersList.find(u => u.username === uInput && u.password === pInput);
            if (foundUser) {
                currentUser = foundUser;
                if (errBox) errBox.classList.add('hidden');
                document.getElementById('loginModal').classList.add('hidden');
                
                // Update Header UI
                document.getElementById('headerUserName').innerText = currentUser.name;
                document.getElementById('headerUserRole').innerText = currentUser.role;
                document.getElementById('userHeaderBadge').classList.remove('hidden');
                document.getElementById('userHeaderBadge').classList.add('flex');

                // Auto-fill Auditor field in form
                if (document.getElementById('formAuditor')) {
                    document.getElementById('formAuditor').value = currentUser.name;
                }

                showToast(`ยินดีต้อนรับคุณ ${currentUser.name}`);
            } else {
                if (errBox) {
                    errBox.classList.remove('hidden');
                    document.getElementById('loginErrorText').innerText = 'ชื่อผู้ใช้หรือรหัสผ่านไม่ถูกต้อง';
                }
            }
        }

        function handleLogout() {
            currentUser = null;
            document.getElementById('userHeaderBadge').classList.add('hidden');
            document.getElementById('userHeaderBadge').classList.remove('flex');
            document.getElementById('loginUsername').value = '';
            document.getElementById('loginPassword').value = '';
            document.getElementById('loginModal').classList.remove('hidden');
            showToast('ออกจากระบบเรียบร้อยแล้ว');
        }

        function setTodayWeek() {
            const date = new Date();
            const d = new Date(Date.UTC(date.getFullYear(), date.getMonth(), date.getDate()));
            const dayNum = d.getUTCDay() || 7;
            d.setUTCDate(d.getUTCDate() + 4 - dayNum);
            const yearStart = new Date(Date.UTC(d.getUTCFullYear(),0,1));
            const weekNo = Math.ceil((((d - yearStart) / 86400000) + 1)/7);
            const weekString = `${d.getUTCFullYear()}-W${weekNo < 10 ? '0' + weekNo : weekNo}`;
            if (document.getElementById('formWeek')) {
                document.getElementById('formWeek').value = weekString;
            }
        }

        function handleImageUpload(input, itemId) {
            if (input.files && input.files[0]) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    itemImages[itemId] = e.target.result;
                    const previewBox = document.getElementById(`img_preview_box_${itemId}`);
                    const previewImg = document.getElementById(`img_preview_${itemId}`);
                    if (previewBox && previewImg) {
                        previewImg.src = e.target.result;
                        previewBox.classList.remove('hidden');
                        previewBox.classList.add('flex');
                    }
                };
                reader.readAsDataURL(input.files[0]);
            }
        }

        function calculateLiveScore() {
            let totalMax = 0;
            let totalEarned = 0;
            let qMax = 0, qEarned = 0;
            let sMax = 0, sEarned = 0;
            let cMax = 0, cEarned = 0;

            checklistState.forEach(item => {
                const selectedRadios = document.getElementsByName(`score_${item.id}`);
                let val = 100;
                for (let r of selectedRadios) {
                    if (r.checked) {
                        val = parseInt(r.value);
                        break;
                    }
                }

                totalMax += 100;
                totalEarned += val;

                if (item.qscPillar === 'Q') { qMax += 100; qEarned += val; }
                else if (item.qscPillar === 'S') { sMax += 100; sEarned += val; }
                else if (item.qscPillar === 'C') { cMax += 100; cEarned += val; }
            });

            const totalScore = totalMax > 0 ? Math.round((totalEarned / totalMax) * 100) : 100;
            const qScore = qMax > 0 ? Math.round((qEarned / qMax) * 100) : 100;
            const sScore = sMax > 0 ? Math.round((sEarned / sMax) * 100) : 100;
            const cScore = cMax > 0 ? Math.round((cEarned / cMax) * 100) : 100;

            document.getElementById('liveTotalScore').innerText = `${totalScore}%`;
            document.getElementById('liveQScore').innerText = `${qScore}%`;
            document.getElementById('liveSScore').innerText = `${sScore}%`;
            document.getElementById('liveCScore').innerText = `${cScore}%`;

            if (document.getElementById('mobileLiveTotalScore')) {
                document.getElementById('mobileLiveTotalScore').innerText = `${totalScore}%`;
                document.getElementById('mobileLiveQScore').innerText = `${qScore}%`;
                document.getElementById('mobileLiveSScore').innerText = `${sScore}%`;
                document.getElementById('mobileLiveCScore').innerText = `${cScore}%`;
            }
        }

        function handleAuditSubmit(e) {
            e.preventDefault();
            const weekVal = document.getElementById('formWeek').value;
            const auditorVal = document.getElementById('formAuditor').value.trim();

            if (!weekVal || !auditorVal) {
                alert('กรุณากรอกข้อมูลสัปดาห์และชื่อผู้ประเมินให้ครบถ้วน');
                return;
            }

            let totalMax = 0, totalEarned = 0;
            let qMax = 0, qEarned = 0;
            let sMax = 0, sEarned = 0;
            let cMax = 0, cEarned = 0;
            let failedItems = [];

            checklistState.forEach(item => {
                const radios = document.getElementsByName(`score_${item.id}`);
                let val = 100;
                for (let r of radios) {
                    if (r.checked) {
                        val = parseInt(r.value);
                        break;
                    }
                }
                const note = document.getElementById(`note_${item.id}`)?.value || '';

                totalMax += 100;
                totalEarned += val;

                if (item.qscPillar === 'Q') { qMax += 100; qEarned += val; }
                else if (item.qscPillar === 'S') { sMax += 100; sEarned += val; }
                else if (item.qscPillar === 'C') { cMax += 100; cEarned += val; }

                if (val < 100) {
                    failedItems.push({
                        id: item.id,
                        title: item.title,
                        category: item.category,
                        qscPillar: item.qscPillar,
                        score: val,
                        note: note,
                        image: itemImages[item.id] || null,
                        actionPlan: null
                    });
                }
            });

            const totalScore = totalMax > 0 ? Math.round((totalEarned / totalMax) * 100) : 100;
            const qScore = qMax > 0 ? Math.round((qEarned / qMax) * 100) : 100;
            const sScore = sMax > 0 ? Math.round((sEarned / sMax) * 100) : 100;
            const cScore = cMax > 0 ? Math.round((cEarned / cMax) * 100) : 100;

            const newRecord = {
                id: 'audit_' + Date.now(),
                week: weekVal,
                branch: 'Central World',
                auditor: auditorVal,
                date: new Date().toLocaleDateString('th-TH'),
                total: totalScore,
                qScore: qScore,
                sScore: sScore,
                cScore: cScore,
                status: totalScore >= 85 ? 'PASSED' : 'ACTION NEEDED',
                failedItems: failedItems
            };

            historyData.unshift(newRecord);

            renderHistoryTable();
            updateDashboardData();
            switchTab('dashboard');
            showToast('บันทึกผลการตรวจประเมินเรียบร้อยแล้ว');
        }

        function handleCapaImageUpload(input) {
            if (input.files && input.files[0]) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    currentCapaImage = e.target.result;
                    document.getElementById('capaImagePreview').src = e.target.result;
                    document.getElementById('capaImagePreviewBox').classList.remove('hidden');
                    document.getElementById('capaImageName').innerText = input.files[0].name;
                };

        function saveActionPlan(e) {
            e.preventDefault();
            const record = historyData.find(r => r.id === currentCapaAuditId);
            if (!record) return;

            const item = record.failedItems.find(i => i.id === currentCapaItemId);
            if (!item) return;

            item.actionPlan = {
                action: document.getElementById('capaAction').value,
                prevention: document.getElementById('capaPrevention').value,
                owner: document.getElementById('capaOwner').value,
                dueDate: document.getElementById('capaDueDate').value,
                image: currentCapaImage
            };

            closeActionModal();
            updateDashboardData();
            showToast('บันทึกแผนติดตามแก้ไขข้อบกพร่อง (CAPA) เรียบร้อย');
        }

        function viewHistoryDetail(idx) {
            const record = historyData[idx];
            if (!record) return;

            document.getElementById('detailTotalScore').innerText = `${record.total}%`;
            document.getElementById('detailQScore').innerText = `${record.qScore}%`;
            document.getElementById('detailSScore').innerText = `${record.sScore}%`;
            document.getElementById('detailCScore').innerText = `${record.cScore}%`;

        function exportHistoryCSV() {
            if (historyData.length === 0) {
                alert('ยังไม่มีข้อมูลสำหรับส่งออก');
                return;
            }
            let csvStr = "\uFEFFสัปดาห์,สาขา,ผู้ประเมิน,Quality(%),Service(%),Cleanliness(%),คะแนนรวม(%),สถานะ\n";
            historyData.forEach(row => {
                csvStr += `"${row.week}","${row.branch}","${row.auditor}",${row.qScore},${row.sScore},${row.cScore},${row.total},"${row.status}"\n`;
            });
            const blob = new Blob([csvStr], { type: 'text/csv;charset=utf-8;' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `xBloom_QSC_Audit_Report_${new Date().toISOString().slice(0,10)}.csv`;
            a.click();
            showToast('ดาวน์โหลดไฟล์ CSV เรียบร้อยแล้ว');
        }

        function handleCreateCriterion(e) {
            e.preventDefault();
            const cat = document.getElementById('modalCategory').value;
            const pillar = document.getElementById('modalPillar').value;
            const title = document.getElementById('modalTitle').value.trim();
            const desc = document.getElementById('modalDesc').value.trim();

            const newItem = {
                id: 'item_' + Date.now(),
                category: cat,
                qscPillar: pillar,
                title: title,
                desc: desc
            };

            checklistState.push(newItem);
            closeAddItemModal();
            renderSettingsList();
            renderAuditForm();
            showToast('เพิ่มหัวข้อประเมินเรียบร้อยแล้ว');
        }

        function openEditItemModal(id) {
            const item = checklistState.find(i => i.id === id);
            if (!item) return;

            document.getElementById('editModalItemId').value = item.id;
            document.getElementById('editModalCategory').value = item.category;
            document.getElementById('editModalPillar').value = item.qscPillar;
            document.getElementById('editModalTitle').value = item.title;
            document.getElementById('editModalDesc').value = item.desc || '';

            document.getElementById('editItemModal').classList.remove('hidden');
        }

        function closeEditItemModal() {
            document.getElementById('editItemModal').classList.add('hidden');
        }

        function handleEditCriterion(e) {
            e.preventDefault();
            const id = document.getElementById('editModalItemId').value;
            const item = checklistState.find(i => i.id === id);
            if (!item) return;

            item.category = document.getElementById('editModalCategory').value;
            item.qscPillar = document.getElementById('editModalPillar').value;
            item.title = document.getElementById('editModalTitle').value.trim();
            item.desc = document.getElementById('editModalDesc').value.trim();

            closeEditItemModal();
            renderSettingsList();
            renderAuditForm();
            showToast('แก้ไขข้อมูลเกณฑ์ประเมินเรียบร้อยแล้ว');
        }

        function deleteCriterion(id) {
            if (confirm('คุณต้องการลบข้อประเมินนี้ใช่หรือไม่?')) {
                checklistState = checklistState.filter(i => i.id !== id);
                renderSettingsList();
                renderAuditForm();
                showToast('ลบข้อประเมินเรียบร้อยแล้ว');
            }
        }

        function showToast(msg) {
            const toast = document.getElementById('toast');
            const toastMsg = document.getElementById('toastMessage');
            if (!toast || !toastMsg) return;

            toastMsg.innerText = msg;
            toast.classList.remove('translate-y-24', 'opacity-0');
            toast.classList.add('translate-y-0', 'opacity-100');

            setTimeout(() => {
                toast.classList.add('translate-y-24', 'opacity-0');
                toast.classList.remove('translate-y-0', 'opacity-100');
            }, 3000);
        }
